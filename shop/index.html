import json import os import requests from jinja2 import Template # ১. মাস্টার ডিজাইন (এটি দিয়েই সব পেজ তৈরি হবে) PAGE_TEMPLATE = """ {{ title }}

{{ title }}

{% for p in products %}

{{ p.name }}

{{ p.price }}

{% endfor %}
Go Back to Home """ # ২. হোমপেজ ডিজাইন (index.html) INDEX_TEMPLATE = """ Affiliate Empire

Our Best Deal Categories

    {% for cat in categories %}
  • {{ cat.name }}
    • {% endfor %}
""" def generate_empire(): # categories.json থেকে নামগুলো লোড করা with open('categories.json', 'r') as f: categories_names = json.load(f) all_pages = [] for name in categories_names: slug = name.lower().replace(" ", "-") print(f"🛠 Generating: {name}") # এখানে আমরা ডামি ডাটা দিচ্ছি (স্ক্র্যাপার পার্ট এখানে যুক্ত হবে) products = [ {"name": f"{name} Product 1", "price": "₹15,000", "image": "https://via.placeholder.com/200", "link": "#"}, {"name": f"{name} Product 2", "price": "₹18,000", "image": "https://via.placeholder.com/200", "link": "#"} ] # ক্যাটাগরি পেজ তৈরি করা html = Template(PAGE_TEMPLATE).render(title=name, products=products) with open(f"{slug}.html", "w", encoding="utf-8") as f: f.write(html) all_pages.append({"name": name, "slug": slug}) # ৩. সবশেষে index.html (Home Page) তৈরি করা index_html = Template(INDEX_TEMPLATE).render(categories=all_pages) with open("index.html", "w", encoding="utf-8") as f: f.write(index_html) if __name__ == "__main__": generate_empire()
